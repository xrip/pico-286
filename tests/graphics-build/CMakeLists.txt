cmake_minimum_required(VERSION 3.21)
# From the repo root, with PICO_SDK_PATH set:
# cmake -S tests/graphics-build -B build/rp2040-pio-smoke -G Ninja -DPICO_TOOLCHAIN_PATH=<toolchain>
# cmake --build build/rp2040-pio-smoke
set(PICO_BOARD pico CACHE STRING "Driver smoke-test board")
set(PICO_PLATFORM rp2040 CACHE STRING "Driver smoke-test platform")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(graphics_smoke C CXX ASM)
set(CMAKE_C_STANDARD 17)
pico_sdk_init()

get_filename_component(PC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(PICO_PLATFORM STREQUAL "rp2350-arm-s")
    set(HDMI_HSTX ON)
    option(NTSC_TV "Run composite with HSTX" ON)
else()
    set(HDMI ON)
endif()
set(GRAPHICS_BUILTIN_FRAMEBUFFER OFF CACHE BOOL "Application owns its VRAM")
if(HDMI_HSTX)
    set(GRAPHICS_NO_CLOCK_SETUP OFF)
else()
    set(GRAPHICS_NO_CLOCK_SETUP ON)
endif()
set(HDMI_BASE_PIN 12 CACHE STRING "First HDMI pin")
add_subdirectory("${PC_ROOT}/drivers/graphics" graphics)
if(HDMI_HSTX)
    add_subdirectory("${PC_ROOT}/drivers/hdmi-hstx" hdmi-hstx)
else()
    add_subdirectory("${PC_ROOT}/drivers/hdmi" hdmi)
endif()
if(NTSC_TV)
    set(NTSC_PIN_OUTPUT 22)
    add_subdirectory("${PC_ROOT}/drivers/ntsc-composite" ntsc-composite)
endif()
add_executable(graphics_smoke main.c)
target_compile_definitions(graphics_smoke PRIVATE
    GRAPHICS_PC_EMULATOR=1 HDMI_BASE_PIN=${HDMI_BASE_PIN}
    PICO_PANIC_FUNCTION= PICO_PRINTF_ALWAYS_INCLUDED=0)
target_compile_options(graphics_smoke PRIVATE -O3 -flto -fno-tree-loop-distribute-patterns)
target_link_options(graphics_smoke PRIVATE -flto -Wl,--print-memory-usage)
target_link_libraries(graphics_smoke PRIVATE graphics pico_stdlib hardware_clocks hardware_vreg)
if(HDMI_HSTX)
    target_link_libraries(graphics_smoke PRIVATE hdmi-hstx)
else()
    target_link_libraries(graphics_smoke PRIVATE hdmi)
endif()
if(NTSC_TV)
    target_link_libraries(graphics_smoke PRIVATE ntsc-composite)
endif()
pico_define_boot_stage2(smoke_boot2 ${PICO_DEFAULT_BOOT_STAGE2_FILE})
target_compile_definitions(smoke_boot2 PRIVATE PICO_FLASH_SPI_CLKDIV=4)
if(HDMI_HSTX)
    target_compile_definitions(smoke_boot2 PRIVATE PICO_FLASH_SPI_RXDELAY=4)
    target_compile_definitions(graphics_smoke PRIVATE PICO_EMBED_XIP_SETUP=1)
endif()
pico_set_boot_stage2(graphics_smoke smoke_boot2)
pico_enable_stdio_usb(graphics_smoke 0)
pico_enable_stdio_uart(graphics_smoke 0)
pico_set_printf_implementation(graphics_smoke none)
pico_add_extra_outputs(graphics_smoke)
