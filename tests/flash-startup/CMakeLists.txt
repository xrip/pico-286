cmake_minimum_required(VERSION 3.21)
set(PICO_BOARD pico2 CACHE STRING "Board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Platform")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(flash_startup C CXX ASM)
set(CMAKE_C_STANDARD 17)
pico_sdk_init()
add_executable(flash_startup main.c)
target_include_directories(flash_startup PRIVATE ../../drivers/flash-config)
option(TEST_EMBED_BOOT "Run an embedded boot stage before main" OFF)
target_compile_definitions(flash_startup PRIVATE PICO_EMBED_XIP_SETUP=$<BOOL:${TEST_EMBED_BOOT}>)
target_compile_options(flash_startup PRIVATE -O2 -fno-jump-tables -fno-tree-loop-distribute-patterns)
target_link_libraries(flash_startup PRIVATE pico_stdlib hardware_clocks hardware_vreg hardware_watchdog)
add_subdirectory(../../drivers/fatfs fatfs)
add_subdirectory(../../drivers/sdcard sdcard)
target_link_libraries(flash_startup PRIVATE fatfs sdcard)
target_compile_definitions(flash_startup PRIVATE
    SDCARD_PIN_SPI0_CS=5 SDCARD_PIN_SPI0_SCK=6 SDCARD_PIN_SPI0_MOSI=7 SDCARD_PIN_SPI0_MISO=4)
option(TEST_SERIAL_BOOT "Test the read-only serial boot stage" OFF)
if(TEST_SERIAL_BOOT)
    pico_define_boot_stage2(serial_boot ${PICO_BOOT_STAGE2_DIR}/boot2_generic_03h.S)
    target_compile_definitions(serial_boot PRIVATE PICO_FLASH_SPI_CLKDIV=12 PICO_FLASH_SPI_RXDELAY=1)
else()
    pico_define_boot_stage2(serial_boot ${PICO_DEFAULT_BOOT_STAGE2_FILE})
    target_compile_definitions(serial_boot PRIVATE PICO_FLASH_SPI_CLKDIV=4 PICO_FLASH_SPI_RXDELAY=4)
endif()
pico_set_boot_stage2(flash_startup serial_boot)
pico_enable_stdio_usb(flash_startup 1)
pico_enable_stdio_uart(flash_startup 0)
pico_add_extra_outputs(flash_startup)
