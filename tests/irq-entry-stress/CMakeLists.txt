cmake_minimum_required(VERSION 3.21)
set(PICO_BOARD pico2 CACHE STRING "Board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Platform")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(irq_entry_stress C CXX ASM)
set(CMAKE_C_STANDARD 17)
pico_sdk_init()
if(NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
    message(FATAL_ERROR "This test needs RP2350 Arm")
endif()
set(TEST_CPU_MHZ 504 CACHE STRING "CPU clock: 504 or 252 MHz")
set(TEST_FLASH_MHZ 84 CACHE STRING "Flash clock: 84 or 42 MHz")
option(TEST_PROBE_RAM "Place the core 1 work loop in SRAM" OFF)
option(TEST_CORE0_RAM "Place the core 0 work loop and lookup data in SRAM" OFF)
option(TEST_HSTX "Use the real HSTX IRQ instead of the timer IRQ" OFF)
option(TEST_PSRAM "Use GPIO8 QMI PSRAM for the core 0 lookup" OFF)
set(TEST_PSRAM_MHZ 126 CACHE STRING "PSRAM clock: 126 or 84 MHz")
if(TEST_PSRAM AND NOT TEST_CORE0_RAM)
    message(FATAL_ERROR "TEST_PSRAM requires TEST_CORE0_RAM=ON")
endif()
if(NOT TEST_PSRAM_MHZ MATCHES "^(126|84)$")
    message(FATAL_ERROR "Use PSRAM 126 or 84 MHz")
endif()
if(NOT TEST_CPU_MHZ MATCHES "^(504|252)$" OR NOT TEST_FLASH_MHZ MATCHES "^(84|42)$")
    message(FATAL_ERROR "Use CPU 504/252 and flash 84/42 MHz")
endif()
get_filename_component(PC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(irq_entry_stress main.c "${PC_ROOT}/src/fault_trace.c")
add_subdirectory("${PC_ROOT}/drivers/flash-config" flash-config)
target_include_directories(irq_entry_stress PRIVATE "${PC_ROOT}/src")
target_compile_definitions(irq_entry_stress PRIVATE
    TEST_CPU_MHZ=${TEST_CPU_MHZ} TEST_FLASH_MHZ=${TEST_FLASH_MHZ}
    TEST_PROBE_RAM=$<BOOL:${TEST_PROBE_RAM}> TEST_CORE0_RAM=$<BOOL:${TEST_CORE0_RAM}>
    TEST_HSTX=$<BOOL:${TEST_HSTX}> TEST_PSRAM=$<BOOL:${TEST_PSRAM}>
    TEST_PSRAM_MHZ=${TEST_PSRAM_MHZ} PICO_EMBED_XIP_SETUP=0
    PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000
    PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=1)
target_compile_options(irq_entry_stress PRIVATE -O2 -g3 -fno-lto -fno-ipa-icf -fno-tree-loop-distribute-patterns)
target_link_options(irq_entry_stress PRIVATE -fno-lto -Wl,--print-memory-usage)
target_link_libraries(irq_entry_stress PRIVATE flash-config pico_stdlib pico_multicore
    hardware_clocks hardware_vreg hardware_timer hardware_exception hardware_irq)
if(TEST_HSTX)
    target_sources(irq_entry_stress PRIVATE "${PC_ROOT}/drivers/hdmi-hstx/hstx_video.c")
    target_include_directories(irq_entry_stress PRIVATE "${PC_ROOT}/drivers/hdmi-hstx")
    target_link_libraries(irq_entry_stress PRIVATE hardware_dma hardware_gpio hardware_pll hardware_resets hardware_sync)
endif()
if(TEST_PROBE_RAM)
    set(PROBE_NAME ram)
else()
    set(PROBE_NAME flash)
endif()
set(TEST_NAME "irq-entry-${TEST_CPU_MHZ}-f${TEST_FLASH_MHZ}-${PROBE_NAME}")
if(TEST_HSTX)
    string(APPEND TEST_NAME "-hstx")
endif()
if(TEST_CORE0_RAM)
    string(APPEND TEST_NAME "-c0ram")
endif()
if(TEST_PSRAM)
    string(APPEND TEST_NAME "-psram${TEST_PSRAM_MHZ}")
endif()
set_target_properties(irq_entry_stress PROPERTIES OUTPUT_NAME "${TEST_NAME}")
pico_enable_stdio_usb(irq_entry_stress 1)
pico_enable_stdio_uart(irq_entry_stress 0)
pico_add_extra_outputs(irq_entry_stress)
